import matplotlib

from fastdtw import fastdtw
from scipy.spatial.distance import euclidean

# --- Ayarlar ---
DOSYA_ADI = "lab_gaze_tracking_data.csv"

def load_data(filepath):
    """Veriyi yükler, temel kontrolleri (Sanity Checks) yapar."""
    if not os.path.exists(filepath):
        raise FileNotFoundError(f"HATA: '{filepath}' bulunamadı.")
    
    df = pd.read_csv(filepath)
    if df.empty:
        raise ValueError("Dosya boş!")
        
    # Zamanın monotonik artıp artmadığını kontrol et
    if not df["Time_s"].is_monotonic_increasing:
        print("UYARI: Time_s sütunu monotonik değil (Zaman atlamaları olabilir)!")
        
    # Ekran boyutlarını CSV'den oku; yoksa veriden tahmin et (fallback)
    if "Screen_W" in df.columns and "Screen_H" in df.columns:
        screen_w = int(df.loc[0, "Screen_W"])
        screen_h = int(df.loc[0, "Screen_H"])
    else:
        screen_w = int(df["Target_X"].max() + 100)
        screen_h = int(df["Target_Y"].max() + 100)
    
    # Temel Hatayı (Raw Error) Hesapla
    df["Raw_Error"] = np.sqrt((df["Target_X"] - df["Gaze_X"])**2 + (df["Target_Y"] - df["Gaze_Y"])**2)
    return df, screen_w, screen_h

def detect_artifacts(df):
    """Çoklu kurala dayalı (Multi-rule) Artefakt Tespiti."""
    df["Artifact_Type"] = "None"
    df["Is_Artifact"] = False
    
    # 1. Klasik Donma (Frame Freeze) Kontrolü
    dx = df["Gaze_X"].diff()
    dy = df["Gaze_Y"].diff()
    freeze_mask = (dx == 0) & (dy == 0)
    
    # 2. Hız Tabanlı (Speed Spike) Outlier Kontrolü (MAD ile Robust Z-Score)
    dt = df["Time_s"].diff().fillna(0.016) # ~60 FPS varsayılan delta
    speed = np.sqrt(dx**2 + dy**2) / dt
    mad = median_abs_deviation(speed.dropna())
    speed_z = 0.6745 * (speed - speed.median()) / (mad + 1e-6)
    spike_mask = speed_z > 5.0 # 5 std sapma dışı ani zıplamalar
    
    # 3. EAR ve Face Detect Kontrolü 
    if "EAR" in df.columns and "Face_Detected" in df.columns:
        # Eğer EAR < thresh ise Göz Kırpma (Blink) olarak işaretle
        ear_mask = df["EAR"] < 0.14
        # Eğer Face_Detected == 0 ise Takip Kaybı (Tracking Lost) olarak işaretle
        face_mask = df["Face_Detected"] == 0
        df.loc[ear_mask, "Artifact_Type"] = "Blink (EAR)"
        df.loc[face_mask, "Artifact_Type"] = "Tracking Lost"
    else:
        ear_mask = pd.Series(False, index=df.index)
        face_mask = pd.Series(False, index=df.index)

    # Artefaktları Birleştir ve İşaretle
    df.loc[freeze_mask & (df["Artifact_Type"] == "None"), "Artifact_Type"] = "UI Freeze"
    df.loc[spike_mask & (df["Artifact_Type"] == "None"), "Artifact_Type"] = "Speed Spike"
    
    # Göz Kırpma, Yüz Kaybı, Donma veya Spike varsa Artefakt kabul et
    df["Is_Artifact"] = ear_mask | face_mask | freeze_mask | spike_mask
    
    # İlk frame diff hesaplanamadığı için False yapıyoruz
    df.loc[0, "Is_Artifact"] = False 
    
    # Loglama
    print("\n--- Artefakt Temizleme Logu ---")
    print(df["Artifact_Type"].value_counts())
    
    return df

def compensate_latency(df):
    """Target ile Gaze arasındaki Phase Lag'i (Zaman Gecikmesini) bulup telafi eder."""
    min_rmse = float('inf')
    best_shift = 0
    
    # Target rotasını -30 ile +30 frame (yaklaşık -0.5sn ile +0.5sn) kaydırarak test et
    for shift in range(-30, 31):
        if shift == 0:
            shift_tx, shift_ty = df["Target_X"], df["Target_Y"]
        else:
            shift_tx, shift_ty = df["Target_X"].shift(shift), df["Target_Y"].shift(shift)
            
        err = np.sqrt((shift_tx - df["Gaze_X"])**2 + (shift_ty - df["Gaze_Y"])**2)
        rmse = np.sqrt((err.dropna()**2).mean())
        
        if rmse < min_rmse:
            min_rmse = rmse
            best_shift = shift
            
    # Optimum kaydırmayı uygula
    df["Target_X_Comp"] = df["Target_X"].shift(best_shift)
    df["Target_Y_Comp"] = df["Target_Y"].shift(best_shift)
    df["Compensated_Error"] = np.sqrt((df["Target_X_Comp"] - df["Gaze_X"])**2 + (df["Target_Y_Comp"] - df["Gaze_Y"])**2)
    
    print(f"\nPhase Lag Analizi: Optimum zaman kaydırması {best_shift} frame olarak bulundu.")
    return df, best_shift

def compute_metrics(df_clean, best_shift):
    """Mean, Median, RMSE, p95, Drift ve DTW metriklerini hesaplar."""
    raw_err = df_clean["Raw_Error"].dropna()
    comp_err = df_clean["Compensated_Error"].dropna()
    
    valid_mask = df_clean["Compensated_Error"].notna() & df_clean["Time_s"].notna()
    times = df_clean.loc[valid_mask, "Time_s"].values
    errors = df_clean.loc[valid_mask, "Compensated_Error"].values
    
    # Drift (Zamanla hata artışı) hesaplama - 1. Derece Polinom Eğimi
    if len(times) > 1:
        slope, intercept = np.polyfit(times, errors, 1)
    else:
        slope, intercept = 0.0, 0.0
        
    # --- Yörünge (Trajectory) Tabanlı DTW Analizi ---
    target_trajectory = df_clean.loc[valid_mask, ["Target_X_Comp", "Target_Y_Comp"]].values
    gaze_trajectory = df_clean.loc[valid_mask, ["Gaze_X", "Gaze_Y"]].values
    
    dtw_score = 0.0
    if len(target_trajectory) > 0 and len(gaze_trajectory) > 0:
        try:
            # Optimal zaman bükülmeli mesafeyi (DTW) hesapla
            dtw_distance, _ = fastdtw(target_trajectory, gaze_trajectory, dist=euclidean)
            # Metriğin yorumlanabilir olması için toplam uzunluğa normalize et
            dtw_score = float(round(dtw_distance / len(gaze_trajectory), 2))
        except Exception as e:
            print(f"UYARI: DTW hesaplanırken hata oluştu: {e}")
            dtw_score = -1.0
            
    # JSON serileştirme hatalarını önlemek için standart Python tiplerine çeviriyoruz
    metrics = {
        "Frames_Total": int(len(df_clean)),
        "Phase_Lag_Frames": int(best_shift),
        "Drift_Slope_px_per_sec": float(round(slope, 3)),
        "DTW_Score_Normalized": dtw_score, # Yeni Yörünge Metriği
        "Raw": {
            "Mean": float(round(raw_err.mean(), 2)),
            "Median": float(round(raw_err.median(), 2)),
            "RMSE": float(round(np.sqrt((raw_err**2).mean()), 2)),
            "p95": float(round(np.percentile(raw_err, 95), 2))
        },
        "Latency_Compensated": {
            "Mean": float(round(comp_err.mean(), 2)),
            "Median": float(round(comp_err.median(), 2)),
            "RMSE": float(round(np.sqrt((comp_err**2).mean()), 2)),
            "p95": float(round(np.percentile(comp_err, 95), 2))
        }
    }
    return metrics, slope, intercept

def make_plots(df, df_clean, metrics, slope, intercept, screen_w, screen_h, out_dir):
    """Lab-Grade standartlarında deterministik grafikler üretir."""
    
    timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    # FIG 1: Gaze Interaction Map
    plt.figure(figsize=(10, 8))
    plt.title(
        "Gaze-Target Etkileşim Haritası (∞ Trajektorisi)\n"
        f"Kompense RMSE: {metrics['Latency_Compensated']['RMSE']} px | Run: {timestamp}"
    )

    plt.plot(
        df["Target_X"], df["Target_Y"],
        color="royalblue", linestyle="--", linewidth=1.5, alpha=0.5,
        label="Teorik ∞ Hedef Rotası"
    )

    plt.plot(
        df_clean["Target_X_Comp"], df_clean["Target_Y_Comp"],
        color="lime", linewidth=2.0,
        label=f"Kompense Hedef (Lag: {metrics['Phase_Lag_Frames']} fr)"
    )

    sc = plt.scatter(
        df_clean["Gaze_X"], df_clean["Gaze_Y"],
        c=df_clean["Compensated_Error"],
        cmap="viridis_r",
        s=14,
        alpha=0.6,
        edgecolors="none",
        label="Göz Etkileşim Noktaları (Gaze)"
    )
    cbar = plt.colorbar(sc)
    cbar.set_label("Anlık Lag-Kompanse Hata (px)")
    
    df_art = df[df["Is_Artifact"]]
    if not df_art.empty:
        plt.scatter(df_art["Gaze_X"], df_art["Gaze_Y"], c="gray", marker="x", s=20, alpha=0.3, label="Artefakt")
        
    plt.xlim(0, screen_w)
    plt.ylim(screen_h, 0)
    plt.gca().set_aspect("equal", adjustable="box")
    plt.legend(loc='upper right', fontsize='small', framealpha=0.8)
    plt.grid(True, linestyle=":", alpha=0.5)
    plt.xlabel("X Koordinatı (px)")
    plt.ylabel("Y Koordinatı (px)")
    plt.tight_layout()
    plt.savefig(os.path.join(out_dir, "fig_trajectory.png"), dpi=300)
    plt.close()

    # FIG 2: Spatial Heatmap
    plt.figure(figsize=(10, 8))
    plt.title(
        "Gaze Spatial Heatmap (∞ Trajektorisi Üzerinde Yoğunluk)\n"
        f"n={len(df_clean)} temiz frame"
    )
    
    hb = plt.hexbin(
        df_clean["Gaze_X"], df_clean["Gaze_Y"],
        gridsize=40, cmap="magma_r", mincnt=1, alpha=0.35
    )
    cbar_hb = plt.colorbar(hb)
    cbar_hb.set_label("Göz Örneği Sayısı (Hexbin)")
    
    sns.kdeplot(
        x=df_clean["Gaze_X"], y=df_clean["Gaze_Y"],
        fill=True, cmap="magma", bw_adjust=1.0,
        thresh=0.05, alpha=0.85, gridsize=220
    )
    
    plt.plot(
        df["Target_X"], df["Target_Y"],
        "w--", linewidth=1.5, alpha=0.7,
        label="Referans ∞ Hedef Rotası"
    )
    
    plt.xlim(0, screen_w)
    plt.ylim(screen_h, 0)
    plt.gca().set_aspect("equal", adjustable="box")
    plt.legend()
    plt.grid(True, linestyle="-", color='white', alpha=0.1) 
    plt.xlabel("X Koordinatı (px)")
    plt.ylabel("Y Koordinatı (px)")
    plt.tight_layout()
    plt.savefig(os.path.join(out_dir, "fig_heatmap.png"), dpi=300)
    plt.close()

    # FIG 3: Error vs Time & Drift
    plt.figure(figsize=(10, 5))
    plt.title(f"Zaman İçinde Hata Analizi (Drift Eğimi: {metrics['Drift_Slope_px_per_sec']} px/sn)")
    plt.plot(df_clean["Time_s"], df_clean["Compensated_Error"], "r.", alpha=0.4, label="Anlık Hata (Compensated)")
    
    trend_y = (df_clean["Time_s"] * slope) + intercept
    plt.plot(df_clean["Time_s"], trend_y, "k--", linewidth=2, label="Drift Trendi")
    
    plt.xlabel("Zaman (s)")
    plt.ylabel("Hata (px)")
    plt.ylim(0, max(300, df_clean["Compensated_Error"].max() + 50))
    plt.legend()
    plt.grid(True, linestyle="--", alpha=0.4)
    plt.savefig(os.path.join(out_dir, "fig_error_time.png"), dpi=300)
    plt.close()
    
    # FIG 4: Error Histogram
    plt.figure(figsize=(10, 5))
    plt.title("Hata Dağılımı (Kuyruk Analizi - p95)")
    sns.histplot(df_clean["Raw_Error"].dropna(), color="gray", alpha=0.5, label="Raw Error", kde=True, bins=40)
    sns.histplot(df_clean["Compensated_Error"].dropna(), color="green", alpha=0.5, label="Latency Compensated Error", kde=True, bins=40)
    
    plt.axvline(metrics["Latency_Compensated"]["p95"], color='r', linestyle='dashed', linewidth=2, label=f"p95 Comp: {metrics['Latency_Compensated']['p95']}px")
    plt.axvline(metrics["Latency_Compensated"]["Median"], color='k', linestyle='solid', linewidth=2, label=f"Median Comp: {metrics['Latency_Compensated']['Median']}px")
    
    plt.xlabel("Hata Miktarı (px)")
    plt.legend()
    plt.tight_layout()
    plt.savefig(os.path.join(out_dir, "fig_error_hist.png"), dpi=300)
    plt.close()

def export_report(df_clean, metrics, out_dir):
    """JSON ve Temizlenmiş CSV dosyalarını dışa aktarır."""
    with open(os.path.join(out_dir, "summary.json"), "w", encoding="utf-8") as f:
        json.dump(metrics, f, indent=4, ensure_ascii=False)
        
    df_clean.to_csv(os.path.join(out_dir, "cleaned_data.csv"), index=False)

def main(csv_path=DOSYA_ADI, out_root="runs"):
    print("Lab-Grade Bilimsel Göz Takip Analiz Boru Hattı Başlatılıyor...\n")
    
    timestamp_str = datetime.now().strftime("%Y%m%d_%H%M%S")
    out_dir = os.path.join(out_root, f"exp_{timestamp_str}")
    os.makedirs(out_dir, exist_ok=True)
    
    try:
        # 1. Load Data
        df, w, h = load_data(csv_path)
        
        # 2. Detect Artifacts
        df = detect_artifacts(df)
        df_clean = df[~df["Is_Artifact"]].copy()
        
        # 3. Compensate Latency (Phase Lag)
        df_clean, best_shift = compensate_latency(df_clean)
        
        # 4. Compute Metrics (DTW Dahil)
        metrics, slope, intercept = compute_metrics(df_clean, best_shift)
        
        # Konsol Özeti
        print("\n================================================")
        print(f"ANALİZ SONUCU (Run: {timestamp_str})")
        print("================================================")
        print(json.dumps(metrics, indent=4))
        print("================================================\n")
        
        # 5. Make Plots & Export
        print("Görselleştirmeler ve JSON raporu oluşturuluyor...")
        make_plots(df, df_clean, metrics, slope, intercept, w, h, out_dir)
        export_report(df_clean, metrics, out_dir)
        
        print(f"BAŞARILI! Tüm analiz paketleri şu klasöre kaydedildi: '{out_dir}'")

    except Exception as e:
        print(f"\nBEKLENMEYEN HATA: {e}")

if __name__ == "__main__":
    import argparse

    parser = argparse.ArgumentParser(description="Lab-grade goz takip analiz boru hatti")
    parser.add_argument(
        "--csv",
        type=str,
        default=DOSYA_ADI,
        help=f"Girdi CSV dosya adi/yolu (varsayilan: {DOSYA_ADI})",
    )
    parser.add_argument(
        "--out-root",
        type=str,
        default="runs",
        help="Deney ciktilari icin kok klasor (varsayilan: runs)",
    )
    args = parser.parse_args()

    main(csv_path=args.csv, out_root=args.out_root)
